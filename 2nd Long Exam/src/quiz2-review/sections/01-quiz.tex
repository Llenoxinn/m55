\section{Quiz 2 as Given}\label{ch:quiz}
% =====================================================================

Transcribed from the photos of the projected quiz (\texttt{Coverage/Quiz2-Photo1--3.jpg}).
Total: $22+13+10=45$ points.

\begin{questions}
\item Consider the curve $C$ formed by the intersection of the surfaces
      \[ x^2+y^2+z^2=25\qquad\text{and}\qquad 3x-4z=0. \]
      \begin{enumerate}[label=(\alph*)]
        \item \textbf{[6 points]} Find a vector-valued function $\vr(t)$ that
              parametrizes $C$, oriented counterclockwise when viewed from above
              the $xy$-plane, starting at $P(4,0,3)$.\\
              \emph{Hint:} An ellipse of the form $\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1$
              can be parametrized by $x=a\cos t$, $y=b\sin t$, where $0\le t\le2\pi$.
        \item \textbf{[4 points]} Find a vector equation of the tangent line to $C$ at $P(4,0,3)$.
        \item \textbf{[5 points]} Find the exact arc length of $C$ from $P(4,0,3)$ to
              $Q(0,5,0)$, following the counterclockwise orientation.
        \item \textbf{[7 points]} Find the curvature of $C$ at $P(4,0,3)$.
      \end{enumerate}
\item Consider the scalar function
      \[ f(x,y)=2x^2y+\ln(x^2+y^2)+\frac{4x}{y}. \]
      Let $P(1,1)$ and $Q(4,5)$ be points in $\R^2$.
      \begin{enumerate}[label=(\alph*)]
        \item \textbf{[7 points]} Find the directional derivative of $f$ at $P$ in the direction from $P$ toward $Q$.
        \item \textbf{[4 points]} Determine the unit vector in the direction in which $f$ increases most rapidly at $P$.
        \item \textbf{[2 points]} Find the maximum rate of increase of $f$ at $P$.
      \end{enumerate}
\end{questions}

\textbf{Instructions for Problems 3--6:} \emph{You may answer any or all of the
following four problems. However, only your TWO (2) highest-scoring answers
will be counted toward your final score. Each problem is worth 5 points, for a
maximum of 10 points from this section.}

\begin{questions}[resume]
\item \textbf{[5 points]} Suppose you start at the point $(0,0,3)$ and move 5 units
      along the curve $\vr(t)=\ip{3\sin t,\;4t,\;3\cos t}$ in the direction of
      increasing $t$. Determine your new position.
\item \textbf{[5 points]} Consider the parabola $y=ax^2$. Find all values of $a$ such
      that the curvature of the parabola at the origin is $4$.\\
      \emph{Note:} Use a vector-valued parametrization and the curvature formula
      for vector functions.
\item \textbf{[5 points]} Find the directions in which the directional derivative of
      $f(x,y)=x^2+y^2+xy$ at the point $(1,1)$ has the value $3$. Express your
      answers as unit vectors.
\item \textbf{[5 points]} Find all points at which the direction of fastest increase
      of the function $f(x,y)=x^2+y^2-2x-4y$ is $\vi+\vj$.
\end{questions}

\begin{remark}[What the quiz tells us about the exam]
\begin{itemize}
\item \textbf{Long problems are multi-part and chained:} a curve of intersection
      $\to$ tangent line $\to$ arc length $\to$ curvature; a function $\to$
      $\Du f$ toward a point $\to$ direction and size of the maximum rate.
      An error in part (a) propagates, so verify (a) before moving on.
\item \textbf{Short problems test one idea each, with a twist:} the arc-length
      parameter, curvature with an unknown constant, solving for directions,
      and solving for points.
\item \textbf{Exact answers} are expected: $\frac{5\pi}{2}$, $\frac{23}{5}$, $\sqrt{82}$, $(3\sin1,4,3\cos1)$.
\end{itemize}
\end{remark}
